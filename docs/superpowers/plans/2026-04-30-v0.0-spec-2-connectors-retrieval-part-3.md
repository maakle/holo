# holo v0.0 Spec #2 — Implementation Plan, Part 3: Allowlist CLI + Slack

> Continues from [`Part 1`](./2026-04-30-v0.0-spec-2-connectors-retrieval.md) → [`Part 2`](./2026-04-30-v0.0-spec-2-connectors-retrieval-part-2.md). Run those first.

**Phases covered:** Phase 5 (allowlist CLI with provider-specific list output + coverage panel for Notion + bot-not-in-channel warning for Slack), Phase 6 (Slack bot OAuth + connector + bot-not-invited handling).

**Why these phases together:** the CLI's `holo allowlist list slack` is the surfacing layer for Slack's permission-model edge case (public channels readable via `channels:history` scope alone; private channels need both an allowlist entry AND `/invite @holo` in the channel itself). The CLI and the Slack connector co-design the `[bot-not-in-channel]` warning contract — when sync hits a `not_in_channel` error, it emits `HOLO_SLACK_BOT_NOT_INVITED` and skips; when the CLI lists, it pre-checks `conversations.info` so the founder sees the same edge cases without having to wait for sync.

```ts
// packages/cli/src/commands/allowlist-add.ts
import { sql } from 'drizzle-orm';
import type { DbClient } from '@holo/db';
import { HoloError } from '@holo/errors';

const PROVIDERS = new Set(['github', 'slack', 'notion'] as const);

export interface RunAllowlistAddInput {
  db: DbClient;
  organizationId: string;
  provider: 'github' | 'slack' | 'notion';
  pattern: string;
  exclude: boolean;
  note?: string;
}

export async function runAllowlistAdd(input: RunAllowlistAddInput): Promise<{ id: string }> {
  if (!PROVIDERS.has(input.provider as 'github')) {
    throw new HoloError('HOLO_INVALID_INPUT', { field: 'provider', value: input.provider });
  }
  const patternKind = input.pattern.includes('*') ? 'glob' : 'exact_id';
  const decision = input.exclude ? 'exclude' : 'include';

  const rows = await input.db.execute<{ id: string }>(sql`
    INSERT INTO connector_allowlists
      (organization_id, provider, pattern, pattern_kind, decision, notes)
    VALUES (${input.organizationId}, ${input.provider}, ${input.pattern},
            ${patternKind}, ${decision}, ${input.note ?? null})
    RETURNING id
  `);
  return { id: rows.rows[0].id };
}

export function attach(allowlist: import('commander').Command, deps: {
  db: DbClient; organizationId: string;
}): void {
  allowlist
    .command('add')
    .argument('<provider>', "'github' | 'slack' | 'notion'")
    .argument('<pattern>', 'glob (e.g. kombo-tech/*) or exact id')
    .option('--exclude', 'register as an exclude pattern')
    .option('--note <text>', 'free-form audit note')
    .action(async (provider: string, pattern: string, opts: { exclude?: boolean; note?: string }) => {
      const { id } = await runAllowlistAdd({
        db: deps.db, organizationId: deps.organizationId,
        provider: provider as 'github', pattern,
        exclude: opts.exclude ?? false, note: opts.note,
      });
      process.stdout.write(`added ${id}\n`);
    });
}
```

```ts
// packages/cli/src/commands/allowlist-remove.ts
import { sql } from 'drizzle-orm';
import type { DbClient } from '@holo/db';
import { HoloError } from '@holo/errors';

export interface RunAllowlistRemoveInput {
  db: DbClient;
  organizationId: string;
  id: string;
}

export async function runAllowlistRemove(input: RunAllowlistRemoveInput): Promise<void> {
  const rows = await input.db.execute(sql`
    DELETE FROM connector_allowlists
    WHERE id = ${input.id} AND organization_id = ${input.organizationId}
    RETURNING id
  `);
  if (rows.rows.length === 0) {
    throw new HoloError('HOLO_NOT_FOUND', { resource: 'connector_allowlist', id: input.id });
  }
}

export function attach(allowlist: import('commander').Command, deps: {
  db: DbClient; organizationId: string;
}): void {
  allowlist
    .command('remove')
    .argument('<id>', 'allowlist row id (uuid)')
    .action(async (id: string) => {
      await runAllowlistRemove({ db: deps.db, organizationId: deps.organizationId, id });
      process.stdout.write(`removed ${id}\n`);
    });
}
```

```ts
// packages/cli/src/commands/allowlist-list-github.ts
import { sql } from 'drizzle-orm';
import Table from 'cli-table3';
import type { DbClient } from '@holo/db';

export interface RenderListGithubInput {
  db: DbClient;
  organizationId: string;
}

export async function renderListGithub(input: RenderListGithubInput): Promise<string> {
  const rows = await input.db.execute<{
    id: string; pattern: string; pattern_kind: string; decision: string;
    notes: string | null; created_at: Date;
  }>(sql`
    SELECT id, pattern, pattern_kind, decision, notes, created_at
    FROM connector_allowlists
    WHERE organization_id = ${input.organizationId} AND provider = 'github'
    ORDER BY created_at ASC
  `);

  if (rows.rows.length === 0) {
    return 'no github allowlist patterns — add one with `holo allowlist add github <pattern>`\n';
  }

  const table = new Table({
    head: ['id', 'pattern', 'kind', 'decision', 'notes', 'created'],
    colWidths: [38, 32, 10, 10, 24, 22],
  });
  for (const r of rows.rows) {
    table.push([
      r.id, r.pattern, r.pattern_kind, r.decision,
      r.notes ?? '', r.created_at.toISOString(),
    ]);
  }
  return table.toString() + '\n';
}

export function attach(allowlistList: import('commander').Command, deps: {
  db: DbClient; organizationId: string;
}): void {
  allowlistList
    .command('github')
    .description('list github allowlist patterns (pure SQL dump; glob expansion happens at sync time)')
    .action(async () => {
      const out = await renderListGithub(deps);
      process.stdout.write(out);
    });
}
```

```ts
import { WebClient } from '@slack/web-api';
import type { DbClient } from '@holo/db';
import { sql } from 'drizzle-orm';

export interface ChannelInfo {
  id: string;
  name: string;
  is_private: boolean;
  is_member: boolean;
}

interface CacheEntry { value: ChannelInfo | { error: 'invalid_auth' | 'channel_not_found' }; expiresAt: number; }

export interface SlackClientHelper {
  fetchInfo(channelId: string): Promise<ChannelInfo | { error: 'invalid_auth' | 'channel_not_found' }>;
}

const TTL_MS = 5 * 60 * 1000;

export async function createSlackClientFromCredential(input: {
  db: DbClient; organizationId: string;
}): Promise<SlackClientHelper> {
  const rows = await input.db.execute<{ access_token: string }>(sql`
    SELECT access_token FROM connector_credentials
    WHERE organization_id = ${input.organizationId} AND provider = 'slack'
    ORDER BY connected_at DESC LIMIT 1
  `);
  if (rows.rows.length === 0) {
    return {
      async fetchInfo() { return { error: 'invalid_auth' }; },
    };
  }
  const client = new WebClient(rows.rows[0].access_token);
  const cache = new Map<string, CacheEntry>();
  return {
    async fetchInfo(channelId: string) {
      const cached = cache.get(channelId);
      if (cached && cached.expiresAt > Date.now()) return cached.value;
      try {
        const res = await client.conversations.info({ channel: channelId });
        if (!res.ok || !res.channel) {
          const err = (res.error === 'invalid_auth' || res.error === 'channel_not_found')
            ? { error: res.error as 'invalid_auth' } : { error: 'invalid_auth' as const };
          cache.set(channelId, { value: err, expiresAt: Date.now() + TTL_MS });
          return err;
        }
        const value: ChannelInfo = {
          id: res.channel.id as string,
          name: (res.channel as { name: string }).name,
          is_private: Boolean((res.channel as { is_private?: boolean }).is_private),
          is_member: Boolean((res.channel as { is_member?: boolean }).is_member),
        };
        cache.set(channelId, { value, expiresAt: Date.now() + TTL_MS });
        return value;
      } catch (err) {
        const code = (err as { data?: { error?: string } }).data?.error;
        const value = code === 'invalid_auth' ? { error: 'invalid_auth' as const } : { error: 'invalid_auth' as const };
        cache.set(channelId, { value, expiresAt: Date.now() + TTL_MS });
        return value;
      }
    },
  };
}
```

```ts
export async function renderListSlack(input: RenderListSlackInput): Promise<string> {
  const rows = await input.db.execute<{
    id: string; pattern: string; decision: string; notes: string | null;
  }>(sql`
    SELECT id, pattern, decision, notes
    FROM connector_allowlists
    WHERE organization_id = ${input.organizationId} AND provider = 'slack'
    ORDER BY created_at ASC
  `);

  if (rows.rows.length === 0) {
    return 'no slack allowlist patterns — add one with `holo allowlist add slack <channel-id>`\n';
  }

  const slack = await createSlackClientFromCredential(input);

  const table = new Table({
    head: ['id', 'channel', 'name', 'state', 'decision', 'notes'],
    colWidths: [38, 14, 18, 24, 10, 22],
  });

  let needsInviteHint = false;
  let revokedSeen = false;

  for (const r of rows.rows) {
    const info = await slack.fetchInfo(r.pattern);
    let state: string;
    let name = '';
    if ('error' in info) {
      state = info.error === 'invalid_auth' ? '[token-revoked]' : '[unknown]';
      revokedSeen ||= info.error === 'invalid_auth';
    } else {
      name = `#${info.name}`;
      if (info.is_private && !info.is_member) {
        state = '[bot-not-in-channel]';
        needsInviteHint = true;
      } else {
        state = '[ok]';
      }
    }
    table.push([r.id, r.pattern, name, state, r.decision, r.notes ?? '']);
  }

  let out = table.toString() + '\n';
  if (needsInviteHint) {
    out += '\nHint: run `/invite @holo` in any channel marked [bot-not-in-channel] in Slack, then re-run.\n';
  }
  if (revokedSeen) {
    out += '\nHint: bot token rejected. Reconnect Slack on /connections to refresh credentials.\n';
  }
  return out;
}
```

```ts
// packages/connectors/src/slack/oauth.ts
import { HoloError } from '@holo/errors';
import type { ConnectorTokens, ExchangeCodeInput } from '../contract';

const SCOPES = [
  'channels:history', 'channels:read',
  'groups:history',   'groups:read',
  'users:read',       'team:read',
];

export function buildSlackAuthorizeUrl(input: {
  clientId: string; redirectUri: string; state: string;
}): string {
  const params = new URLSearchParams({
    client_id: input.clientId,
    scope: SCOPES.join(','),
    redirect_uri: input.redirectUri,
    state: input.state,
  });
  return `https://slack.com/oauth/v2/authorize?${params.toString()}`;
}

export async function exchangeSlackCode(input: ExchangeCodeInput & {
  clientId: string; clientSecret: string;
}): Promise<ConnectorTokens> {
  const params = new URLSearchParams({
    code: input.code,
    client_id: input.clientId,
    client_secret: input.clientSecret,
    redirect_uri: input.redirectUri,
  });
  const res = await fetch('https://slack.com/api/oauth.v2.access', {
    method: 'POST',
    headers: { 'content-type': 'application/x-www-form-urlencoded' },
    body: params.toString(),
  });
  const json = await res.json() as {
    ok: boolean; error?: string;
    access_token?: string; scope?: string;
  };
  if (!json.ok || !json.access_token) {
    throw new HoloError('HOLO_OAUTH_EXCHANGE_FAILED', { upstream: 'slack', error: json.error });
  }
  return { accessToken: json.access_token, scope: json.scope };
}
```

```ts
import { WebClient } from '@slack/web-api';

export interface UserDirectory { resolve(userId: string): string; }

const AUTOMATION_USERNAMES = new Set(['github', 'pagerduty', 'circleci', 'linear']);

export async function buildUserDirectory(client: WebClient): Promise<{
  directory: UserDirectory;
  isBotUser(userId: string): boolean;
}> {
  const map = new Map<string, { realName: string; isBot: boolean }>();
  let cursor: string | undefined;
  do {
    const res = await client.users.list({ cursor, limit: 200 });
    for (const m of res.members ?? []) {
      const isBot = Boolean(m.is_bot) || (m.name ? AUTOMATION_USERNAMES.has(m.name) : false);
      map.set(m.id as string, { realName: (m.real_name as string) ?? (m.name as string) ?? m.id!, isBot });
    }
    cursor = res.response_metadata?.next_cursor || undefined;
  } while (cursor);
  return {
    directory: { resolve(userId) { return map.get(userId)?.realName ?? userId; } },
    isBotUser(userId) { return map.get(userId)?.isBot ?? false; },
  };
}
```

```ts
// packages/connectors/src/slack/sync.ts
import { WebClient, ErrorCode } from '@slack/web-api';
import { sql } from 'drizzle-orm';
import type { DbClient } from '@holo/db';
import { HoloError } from '@holo/errors';
import { resolveAllowlist } from '../shared/allowlist';
import { chunkHash, dedupeAgainstDb } from '../shared/content-hash';
import { slackThreadChunker } from '@holo/chunker/slack-thread';
import { buildUserDirectory } from './user-directory';

export interface RunSlackSyncInput {
  db: DbClient;
  organizationId: string;
  sourceId: string;
  accessToken: string;
  isInitial: boolean;
  enqueueEmbed: (payload: {
    chunks: Array<{ content: string; metadata: Record<string, unknown>; aclSubjects: string[]; kind: 'slack-thread' }>;
    organizationId: string; sourceId: string; channelId: string;
  }) => Promise<void>;
}

const BATCH_SIZE = 50;

export async function runSlackSync(input: RunSlackSyncInput): Promise<{ artifactCount: number; newCursor: Date | null }> {
  const allowlist = await resolveAllowlist({
    db: input.db, organizationId: input.organizationId, provider: 'slack',
  });
  const allowedChannels = allowlist.resolved;
  if (allowedChannels.length === 0) {
    throw new HoloError('HOLO_ALLOWLIST_EMPTY', { provider: 'slack' });
  }

  const client = new WebClient(input.accessToken);
  const { directory, isBotUser } = await buildUserDirectory(client);

  // load existing cursor
  const cursorRows = await input.db.execute<{ metadata: { oldest_per_channel?: Record<string, string> } }>(sql`
    SELECT metadata FROM connector_cursors
    WHERE source_id = ${input.sourceId} AND scope = 'sync'
    LIMIT 1
  `);
  const oldestPerChannel: Record<string, string> = cursorRows.rows[0]?.metadata?.oldest_per_channel ?? {};

  let totalArtifacts = 0;

  for (const channelId of allowedChannels) {
    let info: { name: string; is_private: boolean; is_member: boolean };
    try {
      const res = await client.conversations.info({ channel: channelId });
      if (!res.ok || !res.channel) continue;
      info = {
        name: (res.channel as { name: string }).name,
        is_private: Boolean((res.channel as { is_private?: boolean }).is_private),
        is_member: Boolean((res.channel as { is_member?: boolean }).is_member),
      };
    } catch {
      continue; // bot-not-invited handling lives in Task 6.3 — wraps history call, not info
    }

    const oldest = oldestPerChannel[channelId] ?? '0';
    const chunksForChannel: Array<{ content: string; metadata: Record<string, unknown>; aclSubjects: string[]; kind: 'slack-thread' }> = [];
    let maxTsSeen = oldest;

    let nextCursor: string | undefined;
    do {
      const hist = await client.conversations.history({ channel: channelId, oldest, cursor: nextCursor, limit: 200 });
      for (const msg of hist.messages ?? []) {
        if (msg.bot_id || (msg.user && isBotUser(msg.user))) continue;
        const ts = msg.ts as string;
        if (ts > maxTsSeen) maxTsSeen = ts;
        const threadTs = (msg.thread_ts as string | undefined) ?? ts;

        let parent: { user: string; ts: string; text: string };
        let replies: Array<{ user: string; ts: string; text: string }> = [];
        const isThreadParent = msg.thread_ts === undefined || msg.thread_ts === ts;
        if (!isThreadParent) continue; // a reply that already belongs to a thread we'll process via its parent

        if (msg.reply_count && (msg.reply_count as number) > 0) {
          const repRes = await client.conversations.replies({ channel: channelId, ts: threadTs, limit: 200 });
          const all = (repRes.messages ?? [])
            .filter((m) => !(m.bot_id || (m.user && isBotUser(m.user))))
            .map((m) => ({ user: m.user as string, ts: m.ts as string, text: m.text as string }));
          parent = all[0] ?? { user: msg.user as string, ts, text: msg.text as string };
          replies = all.slice(1);
        } else {
          parent = { user: msg.user as string, ts, text: msg.text as string };
        }

        const chunks = await slackThreadChunker.chunk({
          channelId, channelName: info.name,
          threadTs, parent, replies,
          participantUserIds: [parent.user, ...replies.map((r) => r.user)],
          permalink: `https://slack.com/archives/${channelId}/p${threadTs.replace('.', '')}`,
          userDirectory: directory,
        }, { organizationId: input.organizationId, sourceId: input.sourceId, sourceArtifactId: `slack-thread:${channelId}:${threadTs}` });

        chunksForChannel.push(...chunks.map((c) => ({ ...c, kind: 'slack-thread' as const })));
      }
      nextCursor = hist.response_metadata?.next_cursor || undefined;
    } while (nextCursor);

    // content-hash dedupe
    const hashed = chunksForChannel.map((c) => ({ ...c, hash: chunkHash('slack-thread', c.content) }));
    const newHashes = await dedupeAgainstDb({
      db: input.db, organizationId: input.organizationId, hashes: hashed.map((h) => h.hash),
    });
    const newSet = new Set(newHashes);
    const fresh = hashed.filter((h) => newSet.has(h.hash));

    // enqueue embed jobs in batches
    for (let i = 0; i < fresh.length; i += BATCH_SIZE) {
      await input.enqueueEmbed({
        chunks: fresh.slice(i, i + BATCH_SIZE),
        organizationId: input.organizationId, sourceId: input.sourceId, channelId,
      });
    }

    totalArtifacts += fresh.length;

    // bump per-channel cursor (Slack ts are floats-as-strings; +1µs suffices)
    if (maxTsSeen !== oldest) {
      const bumped = (parseFloat(maxTsSeen) + 0.000001).toFixed(6);
      oldestPerChannel[channelId] = bumped;
    }
  }

  // upsert cursor
  await input.db.execute(sql`
    INSERT INTO connector_cursors (organization_id, source_id, scope, metadata, last_run_at, last_status)
    VALUES (${input.organizationId}, ${input.sourceId}, 'sync',
            ${JSON.stringify({ oldest_per_channel: oldestPerChannel })}::jsonb,
            now(), 'ok')
    ON CONFLICT (source_id, scope) DO UPDATE SET
      metadata = EXCLUDED.metadata, last_run_at = now(), last_status = 'ok'
  `);

  return { artifactCount: totalArtifacts, newCursor: new Date() };
}
```
