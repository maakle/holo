- [ ] **Step 2: Create `eslint-plugin-local/rules/no-bare-throw-error.js`**

```js
/** @type {import('eslint').Rule.RuleModule} */
module.exports = {
  meta: {
    type: 'problem',
    docs: { description: "Disallow bare 'throw new Error(...)'; use holoError() instead." },
    schema: [],
    messages: {
      bareError: "Bare 'throw new Error()' is forbidden in holo. Use holoError() from '@holo/errors'.",
    },
  },
  create(ctx) {
    return {
      ThrowStatement(node) {
        const arg = node.argument;
        if (!arg) return;
        if (arg.type === 'NewExpression' && arg.callee && arg.callee.type === 'Identifier' && arg.callee.name === 'Error') {
          ctx.report({ node, messageId: 'bareError' });
        }
      },
    };
  },
};
```

```js
import tseslint from 'typescript-eslint';
import importX from 'eslint-plugin-import-x';
import local from './eslint-plugin-local/index.js';

export default tseslint.config(
  { ignores: ['**/dist/**', '**/.next/**', '**/node_modules/**', '**/coverage/**', '**/.turbo/**'] },
  ...tseslint.configs.recommended,
  {
    plugins: {
      'import-x': importX,
      local,
    },
    rules: {
      'no-console': 'off',
      '@typescript-eslint/no-explicit-any': 'warn',
      'local/no-bare-throw-error': 'error',
    },
  },
  {
    // Allow bare Error throws inside @holo/errors — that's where HoloError lives
    files: ['packages/errors/**/*.ts'],
    rules: { 'local/no-bare-throw-error': 'off' },
  },
  {
    // Boundary: apps/api and apps/mcp may not import @holo/db directly; only via @holo/retrieval-core
    files: ['apps/api/**/*.ts', 'apps/mcp/**/*.ts'],
    rules: {
      'import-x/no-restricted-paths': [
        'error',
        {
          zones: [
            {
              target: ['apps/api/src/**', 'apps/mcp/src/**'],
              from: ['packages/db/src/**'],
              message: "apps/api and apps/mcp must access the DB via @holo/retrieval-core, not @holo/db. (See ROADMAP.)",
            },
          ],
        },
      ],
    },
  },
);
```

> **Important caveat:** `@holo/auth` and the connector-credentials writes in `apps/web` legitimately need `@holo/db`. The boundary applies to `apps/api` and `apps/mcp` only. In Foundation, `apps/api`'s `AuthGuard` and `apps/mcp`'s `mcp-session.ts` *do* touch the DB through `@holo/db` — that's why the boundary takes effect from spec #2 onward. To avoid breaking Foundation's lint, re-export the necessary subset of `schema` (just `session`, `user`, `organization`) from `@holo/retrieval-core` and import from there, OR add `apps/api/src/auth/**` and `apps/mcp/src/middleware/**` to the rule's `except` list.
>
> **The cleanest fix at this stage:** add a temporary `except` for those two paths and TODO-mark them in `packages/retrieval-core` to absorb in spec #2. Update the rule above to:
>
> ```js
> zones: [
>   {
>     target: ['apps/api/src/**', 'apps/mcp/src/**'],
>     from: ['packages/db/src/**'],
>     except: [
>       'apps/api/src/auth/**',
>       'apps/mcp/src/middleware/**',
>     ],
>     message: '...',
>   },
> ],
> ```

```ts
import { test, expect } from '@playwright/test';
import postgres from 'postgres';

test('GitHub Connect roundtrip flips row to "Connected ✓"', async ({ page, request, context }) => {
  // 1. Seed a session directly in the DB so we don't need to drive Better Auth's OTP UI.
  const sql = postgres(process.env.DATABASE_URL ?? 'postgresql://holo:holo@localhost:5432/holo', { max: 1 });
  const orgRows = await sql`SELECT id FROM organization WHERE slug='default'`;
  const orgId = orgRows[0]!.id as string;
  const userRows = await sql`
    INSERT INTO "user" (email, organization_id, email_verified)
    VALUES ('e2e@example.com', ${orgId}, true)
    ON CONFLICT (email) DO UPDATE SET updated_at = now()
    RETURNING id
  `;
  const userId = userRows[0]!.id as string;
  const token = `e2e-token-${Date.now()}`;
  await sql`
    INSERT INTO "session" (user_id, token, expires_at)
    VALUES (${userId}, ${token}, now() + interval '1 hour')
  `;
  await sql.end();

  await context.addCookies([
    { name: 'better-auth.session_token', value: token, url: 'http://localhost:3000' },
  ]);

  // 2. Mock the upstream GitHub endpoints by intercepting the connector callback's outbound requests.
  await page.route('**://github.com/login/oauth/access_token', async (route) =>
    route.fulfill({
      status: 200,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ access_token: 'gho_e2e_test', scope: 'repo,read:org', token_type: 'bearer' }),
    }),
  );
  await page.route('**://api.github.com/user', async (route) =>
    route.fulfill({
      status: 200,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ id: 999, login: 'e2e-octocat' }),
    }),
  );
  // Intercept the /authorize redirect — fulfill with a redirect back to our callback with code+state.
  await page.route('**://github.com/login/oauth/authorize**', async (route) => {
    const url = new URL(route.request().url());
    const state = url.searchParams.get('state');
    const redirectUri = url.searchParams.get('redirect_uri');
    if (!state || !redirectUri) return route.abort();
    return route.fulfill({
      status: 302,
      headers: { location: `${redirectUri}?code=test_code&state=${encodeURIComponent(state)}` },
    });
  });

  // 3. Drive the UI.
  await page.goto('/connections');
  await expect(page.getByText('GitHub')).toBeVisible();
  await expect(page.getByText('Not connected').first()).toBeVisible();

  await page.getByRole('button', { name: 'Connect' }).first().click();

  // 4. Assert the row flipped.
  await expect(page.getByText('Connected ✓ (e2e-octocat)')).toBeVisible({ timeout: 10_000 });
});
```

> Note: page.route on `https://github.com` from a Next.js server-side request requires Playwright 1.48+'s `page.route` to intercept fetches initiated by the server during the redirect. If interception of server-side fetches doesn't apply in your Playwright version, an alternate pattern is to pass `FETCH_BASE_URL_GITHUB=http://localhost:<mock>/github/...` overrides into `apps/web` and have the connector use those URLs in test mode. The simplest robust path is to add a small mock-github HTTP server (~30 lines) launched as a `globalSetup` and have the connector accept a `baseUrl` override so the OAuth + /user endpoints can be redirected. Implement that override only if the page.route approach doesn't work; otherwise prefer the simpler interception above.
