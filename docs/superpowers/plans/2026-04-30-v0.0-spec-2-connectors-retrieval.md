# holo v0.0 Spec #2 (Connectors + Retrieval) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Wire holo's GitHub OAuth foundation (commit e961750) into a working substrate: ingestion pipeline (clone → chunk → embed → index) for GitHub + Slack + Notion, hybrid retrieval (pgvector + tsvector via RRF), and four MCP tools (search, get_pr, get_thread, get_doc).

**Architecture:** Per-source connectors implement the existing `Connector` interface; per-source chunkers convert raw artifacts → typed Chunks; per-domain embedder router picks `text-embedding-3-large` (prose) or `voyage-code-3` (code), both Matryoshka-truncated to 1024 dims; chunks land in `chunks` table with content-hash dedupe, ACL subjects, and source-specific metadata. BullMQ orchestrates 5 queues with `step()` checkpoint resume. Retrieval is one SQL CTE: vector top-100 + tsvector top-100 → Reciprocal Rank Fusion (k=60) → ACL filter. Four MCP tools wrap retrieval+reassembly for Claude Desktop / Cursor / Cline.

**Tech Stack:** TypeScript, Node 20, Drizzle ORM, Postgres + pgvector, BullMQ, Redis, Octokit, @slack/web-api, @notionhq/client, tree-sitter, OpenAI SDK, Voyage SDK, commander.js, Hono (MCP), Vitest, Playwright.

**Spec:** [`docs/superpowers/specs/2026-04-30-v0.0-spec-2-connectors-design.md`](../specs/2026-04-30-v0.0-spec-2-connectors-design.md)

```ts
import minimatch from 'minimatch';
import { HoloError, HOLO_ALLOWLIST_EMPTY, HOLO_ALLOWLIST_OVERSIZED } from '@holo/errors';
import type { DbClient } from '@holo/db';

export interface ResolveAllowlistInput {
  db: DbClient;
  organizationId: string;
  provider: 'github' | 'slack' | 'notion';
  candidates?: string[];
}

export interface ResolvedAllowlist {
  include: string[];
  exclude: string[];
  resolved: string[];
  matches(candidate: string): boolean;
}

interface Row {
  pattern: string;
  pattern_kind: 'glob' | 'exact_id';
  decision: 'include' | 'exclude';
}

export async function resolveAllowlist(
  input: ResolveAllowlistInput,
): Promise<ResolvedAllowlist> {
  const { db, organizationId, provider, candidates = [] } = input;

  const rows = await db.query<Row>(
    `SELECT pattern, pattern_kind, decision
       FROM connector_allowlists
      WHERE organization_id = $1 AND provider = $2`,
    [organizationId, provider],
  );

  const include = rows.filter((r) => r.decision === 'include').map((r) => r.pattern);
  const exclude = rows.filter((r) => r.decision === 'exclude').map((r) => r.pattern);

  if (include.length === 0) {
    throw new HoloError(HOLO_ALLOWLIST_EMPTY, { provider });
  }

  const matchesAny = (pat: string, kind: 'glob' | 'exact_id', s: string): boolean =>
    kind === 'glob' ? minimatch(s, pat) : pat === s;

  const includeRows = rows.filter((r) => r.decision === 'include');
  const excludeRows = rows.filter((r) => r.decision === 'exclude');

  const resolved = candidates.filter(
    (c) =>
      includeRows.some((r) => matchesAny(r.pattern, r.pattern_kind, c)) &&
      !excludeRows.some((r) => matchesAny(r.pattern, r.pattern_kind, c)),
  );

  // When no candidate set is provided, project the include patterns themselves
  // as the resolved set so the count guard still applies.
  const effective = candidates.length > 0 ? resolved : include;
  if (effective.length > 50) {
    throw new HoloError(HOLO_ALLOWLIST_OVERSIZED, {
      provider,
      count: effective.length,
    });
  }

  return {
    include,
    exclude,
    resolved: effective,
    matches(candidate: string): boolean {
      const inc = includeRows.some((r) =>
        matchesAny(r.pattern, r.pattern_kind, candidate),
      );
      const exc = excludeRows.some((r) =>
        matchesAny(r.pattern, r.pattern_kind, candidate),
      );
      return inc && !exc;
    },
  };
}
```

```ts
// packages/connectors/src/shared/content-hash.ts
import { createHash } from 'node:crypto';
import { sql } from 'drizzle-orm';
import type { DbClient } from '@holo/db';

export function chunkHash(kind: string, content: string): string {
  return createHash('sha256').update(`${kind}:${content}`).digest('hex');
}

export interface DedupeAgainstDbInput {
  db: DbClient;
  organizationId: string;
  hashes: string[];
}

export async function dedupeAgainstDb(input: DedupeAgainstDbInput): Promise<string[]> {
  if (input.hashes.length === 0) return [];

  const rows = await input.db.execute<{ content_hash: string }>(sql`
    SELECT content_hash
    FROM chunks
    WHERE organization_id = ${input.organizationId}
      AND content_hash = ANY(${input.hashes}::text[])
  `);

  const existing = new Set(rows.rows.map((r) => r.content_hash));
  return input.hashes.filter((h) => !existing.has(h));
}
```

```ts
// packages/embedder/src/shared/backoff.ts
import { HoloError } from '@holo/errors';

export interface BackoffOptions {
  maxAttempts?: number;        // default 4
  baseDelayMs?: number;        // default 1000
  upstream: 'openai' | 'voyage';
}

const DEFAULT_DELAYS_MS = [1000, 2000, 4000, 8000];

export async function withBackoff<T>(
  run: () => Promise<T>,
  opts: BackoffOptions,
): Promise<T> {
  const delays = Array.from({ length: opts.maxAttempts ?? 4 }, (_, i) =>
    opts.baseDelayMs ? opts.baseDelayMs * 2 ** i : DEFAULT_DELAYS_MS[i] ?? 8000,
  );
  let lastErr: unknown;
  for (let attempt = 0; attempt <= delays.length; attempt++) {
    try {
      return await run();
    } catch (err) {
      lastErr = err;
      const status = (err as { status?: number }).status;
      const retryable = status === 429 || (status !== undefined && status >= 500 && status < 600);
      if (!retryable || attempt === delays.length) break;
      await new Promise((r) => setTimeout(r, delays[attempt]));
    }
  }
  throw new HoloError('HOLO_INGESTION_RATE_LIMITED', {
    upstream: opts.upstream,
    cause: lastErr,
  });
}
```

```ts
// packages/embedder/src/router.ts
import type { Chunk } from '@holo/chunker';
import type { Embedder, EmbedderRegistry } from './contract';

export function getEmbedderForChunkKind(
  kind: string,
  registry: EmbedderRegistry,
): Embedder {
  return kind === 'github-code' ? registry.voyage : registry.openai;
}

export interface EmbeddedChunk {
  chunk: Chunk & { kind: string };
  embedding: number[];
  embeddingModel: 'openai-3-large' | 'voyage-code-3';
}

export async function embedChunks(
  chunks: (Chunk & { kind: string })[],
  registry: EmbedderRegistry,
): Promise<EmbeddedChunk[]> {
  if (chunks.length === 0) return [];

  // group by chunk kind, preserve input index for re-ordering
  const groups = new Map<string, { indices: number[]; texts: string[] }>();
  chunks.forEach((c, i) => {
    const g = groups.get(c.kind) ?? { indices: [], texts: [] };
    g.indices.push(i);
    g.texts.push(c.content);
    groups.set(c.kind, g);
  });

  const out: EmbeddedChunk[] = new Array(chunks.length);
  await Promise.all(
    [...groups.entries()].map(async ([kind, { indices, texts }]) => {
      const embedder = getEmbedderForChunkKind(kind, registry);
      const vectors = await embedder.embed(texts);
      indices.forEach((origIdx, i) => {
        out[origIdx] = {
          chunk: chunks[origIdx],
          embedding: vectors[i],
          embeddingModel: embedder.model,
        };
      });
    }),
  );

  return out;
}
```
